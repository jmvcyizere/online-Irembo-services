<script>
const user=JSON.parse(localStorage.getItem('registeredUser')||'{}');
document.getElementById('profileName').value=user.name||'';
document.getElementById('profileEmail').value=user.email||'';
document.getElementById('profilePass').value=user.pass||'';

function saveProfile(){
  user.name=document.getElementById('profileName').value;
  user.email=document.getElementById('profileEmail').value;
  user.pass=document.getElementById('profilePass').value;
  localStorage.setItem('registeredUser',JSON.stringify(user));
  alert('Profile saved');
}

function loadTransactions(){
  const txs=JSON.parse(localStorage.getItem('transactions')||'[]').reverse();
  const tbody=document.getElementById('txTable');
  tbody.innerHTML='';
  txs.forEach(t=>{
    const tr=document.createElement('tr');
    tr.className=t.status==='SUCCESS'?'success':'warning';
    tr.innerHTML=`<td>${t.id}</td><td>${t.name}</td><td>${t.phone}</td><td>${t.amount}</td><td>${t.status}</td>
    <td>${t.status==='PENDING'?`<button onclick="confirmPaid('${t.id}')">Confirm</button>`:'✔'}</td>`;
    tbody.appendChild(tr);
  });
}
function confirmPaid(id){
  const txs=JSON.parse(localStorage.getItem('transactions')||'[]');
  const tx=txs.find(t=>t.id===id);
  if(tx){tx.status='SUCCESS';localStorage.setItem('transactions',JSON.stringify(txs));loadTransactions();}
}
loadTransactions();
